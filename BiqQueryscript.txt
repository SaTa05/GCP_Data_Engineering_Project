CREATE TABLE `config_metadata.DataConfig` (
  provider_id           STRING      NOT NULL,
  raw_path              STRING      NOT NULL,
  bronze_path           STRING      NOT NULL,
  silver_path           STRING      NOT NULL,
  file_type             STRING      NOT NULL,   -- csv / json / parquet
  delimiter             STRING,                 -- e.g. ',' | '|' (null for parquet/json)
  write_mode            STRING      NOT NULL,   -- append / overwrite / merge
  header                BOOL        NOT NULL,   -- true if file has header
  ingestion_frequency   STRING      NOT NULL,   -- daily / hourly / ad-hoc
  active                BOOL        NOT NULL,   -- enable/disable provider
  created_at            TIMESTAMP   DEFAULT CURRENT_TIMESTAMP(),
  updated_at            TIMESTAMP
);
